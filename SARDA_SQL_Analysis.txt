
/* The Following query deletes any current table with the following name
*/

DROP TABLE Sarda;

/* The Following query creates the desired table inside the database
*/

CREATE TABLE Sarda (
	Year DateTime2, 
	Sales DECIMAL(10,4), 
	Expenses DECIMAL(10,4), 
	OperatingProfit DECIMAL(10,4), 
	OtherIncome DECIMAL(10,4), 
	Depreciation DECIMAL(10,4), 
	Interest DECIMAL(10,4), 
	Profitbeforetax DECIMAL(10,4), 
	Tax DECIMAL(10,4), 
	Netprofit DECIMAL(10,4), 
	EPS DECIMAL(10,4), 
	Pricetoearning DECIMAL(10,4), 
	Price DECIMAL(10,4)
);

/* The Following query inserts the data in csv format comma delimited to the SQL table
*/

BULK INSERT Sarda
FROM 'C:\Users\Downloads\Sarda Energy_CSV.csv' -- Your Folder Path should be pasted here
WITH (
	FIRSTROW = 2,
	FIELDTERMINATOR = ',',
	ROWTERMINATOR = '\n'

/* The Following query calculates the total Operting profit accumulated yearwise 
and gives a final total at the end
*/

SELECT 
	OperatingProfit AS OP, 
	Year, 
	SUM(Operatingprofit) OVER (
		ORDER BY Year
		ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
		) AS RunningTotal
FROM Sarda;
